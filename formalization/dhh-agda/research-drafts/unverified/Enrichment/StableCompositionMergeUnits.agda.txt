private variable ℓ : Level

module _ (G H : Graph ℓ) where
  private
    A : CubicalSet ℓ
    A = N∞ (HomGraph G H)
    U : CubicalSet ℓ
    U = DayUnitAt ℓ
    HH : CubicalSet ℓ
    HH = N∞ (HomGraph H H)
    GG : CubicalSet ℓ
    GG = N∞ (HomGraph G G)
    abstract
      leftUnitStage : (p q : ℕ) (v : Cell U p) (k : ℕ) (b : Cell (Stage (HomGraph G H) k) q)
        → mergeN∞ G H H p q (mapCells (stableIdentity H) p v)
            (Stable.point (HomGraph G H) q (k , b))
          ≡ reindex A (lower v ⊗c cubeId q) (Stable.point (HomGraph G H) q (k , b))
      leftUnitStage p q v k b =
        let identityAtK : Cell HH p
            identityAtK = Stable.point (HomGraph H H) p
              (k , constantHom (HomGraph H H) (idHom H))
            inputAtK : Cell A q
            inputAtK = Stable.point (HomGraph G H) q (k , b)
            composed : Cell (Stage (HomGraph G H) k) (p + q)
            composed = combineAt G H H k p q
              (constantHom (HomGraph H H) (idHom H)) b
            collapsed : Cell (Stage (HomGraph G H) k) (p + q)
            collapsed = reindex (Stage (HomGraph G H) k) (lower v ⊗c cubeId q) b
            e1 : mergeN∞ G H H p q (mapCells (stableIdentity H) p v) inputAtK
              ≡ mergeN∞ G H H p q identityAtK inputAtK
            e1 = congS {A = Cell HH p} {B = Cell A (p + q)}
              (λ a → mergeN∞ G H H p q a inputAtK)
              (stableIdentity-at-stage H k p v)
            e2 : mergeN∞ G H H p q identityAtK inputAtK
              ≡ Stable.point (HomGraph G H) (p + q) (k , composed)
            e2 = mergeN∞-same-stage-abstract G H H k p q
              (constantHom (HomGraph H H) (idHom H)) b
            e3 : Stable.point (HomGraph G H) (p + q) (k , composed)
              ≡ Stable.point (HomGraph G H) (p + q) (k , collapsed)
            e3 = congS {A = Cell (Stage (HomGraph G H) k) (p + q)} {B = Cell A (p + q)}
              (λ c → Stable.point (HomGraph G H) (p + q) (k , c))
              (composeGrid-left-unit (suc k) p q {G = G} {H = H} (lower v) b)
        in e1 ∙ e2 ∙ e3

      rightUnitStage : (p q : ℕ) (v : Cell U q) (k : ℕ) (a : Cell (Stage (HomGraph G H) k) p)
        → mergeN∞ G G H p q (Stable.point (HomGraph G H) p (k , a))
            (mapCells (stableIdentity G) q v)
          ≡ reindex A ((cubeId p ⊗c lower v) ⋆c cubeRightUnit p)
            (Stable.point (HomGraph G H) p (k , a))
      rightUnitStage p q v k a =
        let identityAtK : Cell GG q
            identityAtK = Stable.point (HomGraph G G) q
              (k , constantHom (HomGraph G G) (idHom G))
            inputAtK : Cell A p
            inputAtK = Stable.point (HomGraph G H) p (k , a)
            composed : Cell (Stage (HomGraph G H) k) (p + q)
            composed = combineAt G G H k p q a
              (constantHom (HomGraph G G) (idHom G))
            collapsed : Cell (Stage (HomGraph G H) k) (p + q)
            collapsed = reindex (Stage (HomGraph G H) k)
              ((cubeId p ⊗c lower v) ⋆c cubeRightUnit p) a
            e1 : mergeN∞ G G H p q inputAtK (mapCells (stableIdentity G) q v)
              ≡ mergeN∞ G G H p q inputAtK identityAtK
            e1 = congS {A = Cell GG q} {B = Cell A (p + q)}
              (mergeN∞ G G H p q inputAtK)
              (stableIdentity-at-stage G k q v)
            e2 : mergeN∞ G G H p q inputAtK identityAtK
              ≡ Stable.point (HomGraph G H) (p + q) (k , composed)
            e2 = mergeN∞-same-stage-abstract G G H k p q a
              (constantHom (HomGraph G G) (idHom G))
            e3 : Stable.point (HomGraph G H) (p + q) (k , composed)
              ≡ Stable.point (HomGraph G H) (p + q) (k , collapsed)
            e3 = congS {A = Cell (Stage (HomGraph G H) k) (p + q)} {B = Cell A (p + q)}
              (λ c → Stable.point (HomGraph G H) (p + q) (k , c))
              (composeGrid-right-unit (suc k) p q {G = G} {H = H} (lower v) a)
        in e1 ∙ e2 ∙ e3

  abstract
    mergeN∞-left-unit : (p q : ℕ) (v : Cell (DayUnitAt ℓ) p) (b : Cell A q)
      → mergeN∞ G H H p q (mapCells (stableIdentity H) p v) b
        ≡ reindex A (lower v ⊗c cubeId q) b
    mergeN∞-left-unit p q v = SQ.elimProp
      {A = Stable.Tagged (HomGraph G H) q} {R = Stable.Step (HomGraph G H) q}
      {P = λ b → mergeN∞ G H H p q (mapCells (stableIdentity H) p v) b
        ≡ reindex A (lower v ⊗c cubeId q) b}
      (λ _ → snd (cells A (p + q)) _ _)
      (λ { (k , b) → leftUnitStage p q v k b })

    mergeN∞-right-unit : (p q : ℕ) (a : Cell A p) (v : Cell (DayUnitAt ℓ) q)
      → mergeN∞ G G H p q a (mapCells (stableIdentity G) q v)
        ≡ reindex A ((cubeId p ⊗c lower v) ⋆c cubeRightUnit p) a
    mergeN∞-right-unit p q a v = SQ.elimProp
      {A = Stable.Tagged (HomGraph G H) p} {R = Stable.Step (HomGraph G H) p}
      {P = λ a → mergeN∞ G G H p q a (mapCells (stableIdentity G) q v)
        ≡ reindex A ((cubeId p ⊗c lower v) ⋆c cubeRightUnit p) a}
      (λ _ → snd (cells A (p + q)) _ _)
      (λ { (k , a) → rightUnitStage p q v k a }) a
