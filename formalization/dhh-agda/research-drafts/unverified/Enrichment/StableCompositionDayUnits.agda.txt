private variable ℓ : Level

module _ (G H : Graph ℓ) where
  private
    A : CubicalSet ℓ
    A = N∞ (HomGraph G H)
    U : CubicalSet ℓ
    U = DayUnitAt ℓ
    HH : CubicalSet ℓ
    HH = N∞ (HomGraph H H)
    GG : CubicalSet ℓ
    GG = N∞ (HomGraph G G)
  private
    composeLeft : CubicalNatural (DayTensor HH A) A
    composeLeft = actualStableComposition G H H
    composeRight : CubicalNatural (DayTensor A GG) A
    composeRight = actualStableComposition G G H
    leftUnitRoute : CubicalNatural (DayTensor U A) A
    leftUnitRoute = dayMap {A = U} {A' = HH} {B = A} {B' = A}
      (stableIdentity H) (cubicalId A) ⋆n composeLeft
    rightUnitRoute : CubicalNatural (DayTensor A U) A
    rightUnitRoute = dayMap {A = A} {A' = A} {B = U} {B' = GG}
      (cubicalId A) (stableIdentity G) ⋆n composeRight

  abstract
    actualStableComposition-left-unit :
      dayMap (stableIdentity H) (cubicalId A) ⋆n actualStableComposition G H H
        ≡ dayLeftUnitAt A
    actualStableComposition-left-unit = dayTensorExt {A = U} {B = A} {C = A}
      leftUnitRoute (dayLeftUnitAt A) λ p q v b →
      cong (mapCells {A = DayTensor HH A} {B = A} composeLeft (p + q))
        (dayMap-β {A = U} {A' = HH} {B = A} {B' = A}
          (stableIdentity H) (cubicalId A) p q v b)
      ∙ actualStableComposition-β G H H p q (mapCells (stableIdentity H) p v) b
      ∙ actualMergeN∞-left-unit G H p q v b
      ∙ sym (dayLeftUnitAt-β A p q v b)

    actualStableComposition-right-unit :
      dayMap (cubicalId A) (stableIdentity G) ⋆n actualStableComposition G G H
        ≡ dayRightUnitAt A
    actualStableComposition-right-unit = dayTensorExt {A = A} {B = U} {C = A}
      rightUnitRoute (dayRightUnitAt A) λ p q a v →
      cong (mapCells {A = DayTensor A GG} {B = A} composeRight (p + q))
        (dayMap-β {A = A} {A' = A} {B = U} {B' = GG}
          (cubicalId A) (stableIdentity G) p q a v)
      ∙ actualStableComposition-β G G H p q a (mapCells (stableIdentity G) q v)
      ∙ actualMergeN∞-right-unit G H p q a v
      ∙ sym (dayRightUnitAt-β A p q a v)


  abstract
    stableComposition-left-unit :
      dayMap {A = U} {A' = HH} {B = A} {B' = A}
        (stableIdentity H) (cubicalId A) ⋆n stableComposition G H H
        ≡ dayLeftUnitAt A
    stableComposition-left-unit =
      cong (λ (f : CubicalNatural (DayTensor HH A) A) →
        _⋆n_ {A = DayTensor U A} {B = DayTensor HH A} {C = A}
          (dayMap {A = U} {A' = HH} {B = A} {B' = A}
            (stableIdentity H) (cubicalId A)) f)
        (stableComposition-actual G H H)
      ∙ actualStableComposition-left-unit

    stableComposition-right-unit :
      dayMap {A = A} {A' = A} {B = U} {B' = GG}
        (cubicalId A) (stableIdentity G) ⋆n stableComposition G G H
        ≡ dayRightUnitAt A
    stableComposition-right-unit =
      cong (λ (f : CubicalNatural (DayTensor A GG) A) →
        _⋆n_ {A = DayTensor A U} {B = DayTensor A GG} {C = A}
          (dayMap {A = A} {A' = A} {B = U} {B' = GG}
            (cubicalId A) (stableIdentity G)) f)
        (stableComposition-actual G G H)
      ∙ actualStableComposition-right-unit
